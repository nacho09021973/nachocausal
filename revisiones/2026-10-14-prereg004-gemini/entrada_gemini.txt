## 4. Observable and estimator

- **Observable:** for each minimal element i of the order, `O(i) = |future(i)|` (future
  cardinality). Counting is legitimate (`Order + Number`); the estimator sees only the order.
- **Split:** sealed 1-D 2-means on O; `improvement` is the split's variance reduction.
- **Gate:** abstain (`sep → 0`, no boundary claimed) iff `improvement < τ(n)`, n = number of
  minimal elements.
- **Bracket (scoring only):** `[max r over low-O minimals, min r over high-O minimals]`;
  `covers` iff `r_lo ≤ 2M ≤ r_hi`; `clean` iff width ≥ 0; point estimate = midpoint.
- **Guard-v:** `verify_order_only` on every causet; it raises on failure (aborting the run).

# =============================================================================
# Frozen statistical primitives (as 1+1, with the 2+1 constants).

def signflip_perm_p(d) -> float:
    """One-sided exact paired sign-flip p-value for mean(d) > 0 (as nachocausal.validate)."""
    d = np.asarray(d, float)
    d = d[np.isfinite(d)]
    n = d.size
    if n == 0:
        return float("nan")
    obs = d.mean()
    if n <= thresholds.PERM_EXACT_MAX_N:
        idx = np.arange(1 << n)
        bits = (idx[:, None] >> np.arange(n)[None, :]) & 1
        means = ((1 - 2 * bits) * d[None, :]).mean(axis=1)
    else:
        rng = np.random.default_rng(0)
        signs = 1 - 2 * rng.integers(0, 2, size=(thresholds.PERM_RANDOM_SAMPLES, n))
        means = (signs * d[None, :]).mean(axis=1)
    return float(np.mean(means >= obs - 1e-12))


def _iqr(x) -> float:
    x = np.asarray(x, float)
    x = x[np.isfinite(x)]
    if x.size < 2:
        return float("nan")
    q75, q25 = np.percentile(x, [75, 25])
    return float(q75 - q25)

def check_seed(seed: int) -> None:

    is in the 2+1 exploration band, or is one of the drawn VALIDATION_SEEDS after the freeze.
    Virgin-band seeds can therefore never be generated before C1/C2, and burned Phase-A seeds
    (outside the band) never at all."""

def generate(seed: int, rho: float) -> dict:
    check_seed(seed)
    rng = np.random.default_rng(seed)
    t, r, phi = generator.sprinkle(thresholds.n_mean(rho), rng)
    reasons: List[str] = []
    stats: dict = {}
    try:
        bh, stats = generator.bh_order(t, r, phi, rng=rng, audit=thresholds.AUDIT_PAIRS)
    except AssertionError as exc:            # fine shooting missed
        return {"t": t, "r": r, "phi": phi, "bh": None, "mk": None, "stats": {},
                "invalid": True, "reasons": [f"generator_assertion: {exc}"]}
    mk = generator.mink_order(t, r, phi)
    for part, a in stats.get("audit_vs_fine", {}).items():
        if a["endpoint_failures"] or a["disagreements_among_hit"]:
            reasons.append(f"audit_{part}: {a}")
    tv_bh = generator.transitivity_violations(bh)
    tv_mk = generator.transitivity_violations(mk)
    mism, _ = generator.radial_check(t, r)
    stats.update(transitivity_violations_BH=tv_bh, transitivity_violations_MINK=tv_mk,
                 radial_check_mismatches=mism)
    if tv_bh:
        reasons.append(f"transitivity_BH={tv_bh}")
    if tv_mk:
        reasons.append(f"transitivity_MINK={tv_mk}")
    if mism:
        reasons.append(f"radial_check_mismatches={mism}")
    return {"t": t, "r": r, "phi": phi, "bh": bh, "mk": mk, "stats": stats,
            "invalid": bool(reasons), "reasons": reasons}

# Per-seed blind phase (order only) -> then reveal r to score.

def per_seed(seed: int, rho: float, guard: bool = True) -> dict:
    t0 = time.time()
    g = generate(seed, rho)
    row: Dict[str, object] = {"seed": int(seed), "rho": float(rho), "N": int(g["t"].size),
                              "invalid": g["invalid"], "invalid_reasons": g["reasons"],
                              "generator_stats": g["stats"]}
    if g["invalid"]:
        row["seconds"] = round(time.time() - t0, 1)
        return row
    blind = {kind: estimator.estimate(order, guard=guard, guard_seed=int(seed))
             for kind, order in (("BH", g["bh"]), ("MINK", g["mk"]))}
    # thr is frozen from O alone for both kinds; ONLY NOW is r revealed (BH only).
    bh, mk = blind["BH"], blind["MINK"]
    if bh["abstained"]:
        br = dict(valid=False, r_lo=float("nan"), r_hi=float("nan"), width=float("nan"),
                  midpoint=float("nan"), covers=False, clean=False)
    else:
        br = blind_bracket(bh["O_by_min"], bh["min_idx"], bh["thr"], g["r"])
    row.update(sep_BH=bh["sep"], sep_MINK=mk["sep"], d=bh["sep"] - mk["sep"],
               abstained_BH=bh["abstained"], abstained_MINK=mk["abstained"],
               n_minimal_BH=bh["n_minimal"], n_minimal_MINK=mk["n_minimal"],
               improvement_BH=bh["improvement"], improvement_MINK=mk["improvement"], **br)
    row["seconds"] = round(time.time() - t0, 1)
    return row


def summarise_level(rows: List[dict], rho: float) -> dict:
    """Level statistics from per-seed rows (pure function: used by run_level and by the
    step-4 can-fail test on synthetic rows)."""
    ok = [r for r in rows if not r["invalid"]]
    res = {
        "rho": float(rho),
        "n_mean": thresholds.n_mean(rho),
        "n_seeds": len(rows),
        "n_invalid": len(rows) - len(ok),
        "n_generator_valid": len(ok),
        "n_abstain_BH": int(sum(bool(r["abstained_BH"]) for r in ok)),
        "n_escape_MINK": int(sum(not r["abstained_MINK"] for r in ok)),
        "theta_loc": thresholds.theta_loc(rho),
        "theta_stab": thresholds.theta_stab(rho),
        "slack_over_2M": thresholds.convergence_slack(rho),
    }
    if len(ok) < thresholds.MIN_VALID_SEEDS:
        res["status"] = "inconclusive"
        return res
    res["p_perm"] = signflip_perm_p([r["d"] for r in ok])
    res["significant"] = bool(res["p_perm"] <= thresholds.P_PERM_THRESHOLD
                              and res["n_abstain_BH"] < thresholds.BH_ABSTAIN_FAIL)
    brk = [r for r in ok if r["valid"]]
    clean = [r for r in brk if r["clean"]]
    widths = np.array([r["width"] / thresholds.TWO_M for r in clean])
    res["n_bracket_valid"] = len(brk)
    res["n_clean"] = len(clean)
    res["median_width_over_2M"] = float(np.median(widths)) if widths.size else float("nan")
    res["iqr_width_over_2M"] = _iqr(widths)
    res["loc_pass"] = bool(res["median_width_over_2M"] <= res["theta_loc"])
    res["loc_inconclusive"] = bool(res["iqr_width_over_2M"] > res["theta_loc"])
    res["coverage_count"] = int(sum(bool(r["covers"]) for r in brk))
    res["coverage_frac"] = res["coverage_count"] / len(brk) if brk else float("nan")
    mids = np.array([r["midpoint"] for r in clean])
    res["boundary_r_std"] = float(np.std(mids)) if mids.size else float("nan")
    res["stab_pass"] = bool(res["boundary_r_std"] <= res["theta_stab"])
    res["escape_pass"] = bool(res["n_escape_MINK"] <= thresholds.MAX_MINK_ESCAPES)
    res["status"] = "scored"
    return res

def run_level(seeds, rho: float, guard: bool = True, progress: Optional[str] = None) -> dict:

def verdict_from_levels(levels: Dict[float, dict], *, guard_ran: bool) -> dict:
    """Frozen-form verdict from level summaries (pure; exercised by the can-fail test)."""
    prim = levels[thresholds.PRIMARY_RHO]
    if prim.get("status") != "scored":
        return {"verdict": "INCONCLUSIVE",
                "reason": f"primary rho={thresholds.PRIMARY_RHO} has fewer than "
                          f"{thresholds.MIN_VALID_SEEDS} generator-valid seeds"}
    required = [rho for rho in thresholds.RHO_GRID if rho >= thresholds.SIGNIFICANCE_MIN_RHO]
    sig_ok = all(levels[rho].get("significant", False) for rho in required)
    slack_ok = True
    grid = list(thresholds.RHO_GRID)
    for a_rho, b_rho in zip(grid[:-1], grid[1:]):
        a, b = levels[a_rho], levels[b_rho]
        if a.get("status") != "scored" or b.get("status") != "scored":
            slack_ok = False
            break
        if b["median_width_over_2M"] > a["median_width_over_2M"] + a["slack_over_2M"]:
            slack_ok = False
    checks = {
        "i_significant_every_rho_ge_min": bool(sig_ok and prim["significant"]),
        "ii_localisation_primary": bool(prim["loc_pass"] and not prim["loc_inconclusive"]
                                        and prim["coverage_frac"] >= thresholds.COVERAGE_FLOOR),
        "ii_convergence_slack": bool(slack_ok),
        "iii_stability_primary": bool(prim["stab_pass"]),
        "iv_flat_escapes_primary": bool(prim["escape_pass"]),
        # (v) Guard-v ran on every causet (it RAISES on failure, aborting the run); a run with
        # the guard off can never pass (v) (auditor report 043, finding 1).
        "v_order_only": bool(guard_ran),
    }
    return {"verdict": "PASS" if all(checks.values()) else "FAIL", "checks": checks}


# =============================================================================
# Full run -> verdict.
# =============================================================================
def run(seeds=None, label: str = VALIDATION_LABEL, guard: bool = True, write: bool = True,
        t_edge: Optional[float] = None) -> dict:
    thresholds.assert_environment()
    seeds = list(thresholds.VALIDATION_SEEDS if seeds is None else seeds)
    if not seeds:
        raise RuntimeError("no seeds: VALIDATION_SEEDS are not drawn (step C2 of comité 052).")
    for s in seeds:
        check_seed(s)
    if label != VALIDATION_LABEL and set(seeds) & set(thresholds.VALIDATION_SEEDS):
        raise RuntimeError("validation seeds may only be run under the validation label.")
    if label == VALIDATION_LABEL:
        if not guard:
            raise RuntimeError("the validation run requires Guard-v (guard=True).")
        if thresholds.STATUS != "FROZEN" or tuple(seeds) != tuple(thresholds.VALIDATION_SEEDS):
            raise RuntimeError("the validation label is reserved for the single blind run on the "
                               "frozen VALIDATION_SEEDS after C1/C2.")
        if os.path.exists(_result_path(label)):
            raise RuntimeError(f"{_result_path(label)} exists: the blind run is one-shot.")
        thresholds.assert_blind_threads()
    t_edge = thresholds.T_EDGE if t_edge is None else float(t_edge)
    if t_edge < thresholds.T_EDGE_MIN:
        out = {"label": label, "seeds": seeds, "t_edge": t_edge, "verdict": "OUT_OF_DOMAIN",
               "reason": f"t_edge={t_edge} < T_EDGE_MIN={thresholds.T_EDGE_MIN} (not a FAIL)"}
        if write:
            write_atomic(out, label)
        return out
    if t_edge != generator.T_EDGE:
        raise RuntimeError("t_edge must equal the generator's literal T_EDGE (no T knob).")
    progress = _progress_path(label) if write else None
    if progress is not None:
        append_progress(progress, {"event": "start", "label": label, "seeds": seeds,
                                   "time": time.strftime("%Y-%m-%dT%H:%M:%S%z")})
    levels = {rho: run_level(seeds, rho, guard, progress) for rho in thresholds.RHO_GRID}
    out = {"label": label, "status_of_thresholds": thresholds.STATUS, "seeds": seeds,
           "t_edge": t_edge, "guard": bool(guard), "levels": levels,
           **verdict_from_levels(levels, guard_ran=guard)}
    if write:
        write_atomic(out, label)
    return out


def _result_path(label: str) -> str:
    return os.path.join(RESULTS_DIR, f"{label}.json")


def _progress_path(label: str) -> str:
    return os.path.join(RESULTS_DIR, f"{label}.progress.jsonl")


def write_atomic(obj: dict, label: str) -> str:
    """Write results/<label>.json atomically: temp file in the same dir, fsync, rename."""
    os.makedirs(RESULTS_DIR, exist_ok=True)
    path = _result_path(label)
    fd, tmp = tempfile.mkstemp(prefix=f".{label}.", suffix=".tmp", dir=RESULTS_DIR)
    try:
        with os.fdopen(fd, "w") as f:
            json.dump(obj, f, indent=2, default=_json_default)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
    except BaseException:
        if os.path.exists(tmp):
            os.unlink(tmp)
        raise
    return path


def _json_default(x):
    if isinstance(x, (np.integer,)):
        return int(x)
    if isinstance(x, (np.floating,)):
        return float(x)
    if isinstance(x, np.ndarray):
        return x.tolist()
    return str(x)

"""ESTIMATOR — order-only, 2+1 bench (CANDIDATE, not frozen).

A thin wrapper around the SEALED estimator-v2 primitives of the 1+1 package, imported UNCHANGED
(comité 052, reproducibility brief): estimate_O_volume (O(i) = |future(i)| over the minimal
elements), two_means_split, improvement, verify_order_only (Guard-v). The 2+1 generator returns
order[i, j] = "i precedes j"; the sealed primitives take the past matrix C[i, j] = "j precedes i",
so C = order.T. Same observable and split as dev/explore_2p1_estimator.py.

Input: the boolean order matrix ONLY. This module imports nothing from nachocausal2p1.scoring
and never sees coordinates (tests/test_2p1_package.py enforces both).
"""

from __future__ import annotations

from typing import Dict, List

import numpy as np

from nachocausal import estimator as sealed

from . import gate


def past_matrix(order: np.ndarray) -> np.ndarray:
    """C[i, j] = j precedes i, from order[i, j] = i precedes j."""
    return np.ascontiguousarray(np.asarray(order, dtype=bool).T)


def estimate(order: np.ndarray, guard: bool = True, guard_seed: int = 0) -> Dict[str, object]:
    """Order-only estimate. Guard-v (sealed verify_order_only on the production observable)
    RAISES if O is not invariant under a pure relabelling. Returns the minimal elements, O,
    the frozen blind threshold, sep, improvement, n and the gate decision."""
    C = past_matrix(order)
    if guard:
        sealed.verify_order_only(C, seed=guard_seed)
    O_by_min, min_idx = sealed.estimate_O_volume(C)
    vals: List[int] = [O_by_min[i] for i in min_idx]
    n = len(min_idx)
    imp = sealed.improvement(vals) if n >= 2 else 0.0
    thr, sep = sealed.two_means_split(vals)
    abstained = gate.abstains(imp, n)
    if abstained:
        sep = 0.0                      # no boundary claimed
    return {"O_by_min": O_by_min, "min_idx": min_idx, "thr": thr, "sep": float(sep),
            "improvement": float(imp), "n_minimal": n, "abstained": bool(abstained)}

"""GATE — estimator-v2 tau(n) abstaining gate for the 2+1 bench (CANDIDATE, not frozen).

Same semantics as the sealed nachocausal.gate: abstain (no boundary claimed) iff
improvement(O_min) < tau(n), n = number of minimal elements; n < 2 abstains.
The table is fixtures/tau_table_2p1.json: scripts/gen_tau_table.build_tau UNCHANGED, one shared
rng, n ascending from 2 to 256, 40000 reps; n <= 128 is bit-equal to the sealed 1+1 table
(comité 052 §9 step 1; dev/PREREG004_STEPS_1_2_2026-10-13.md). The file's SHA256 must equal
thresholds.TAU_FIXTURE_SHA256. n > 256 RAISES: the run aborts, it never extrapolates.
Order-only; imports nothing from the scoring subpackage.
"""

from __future__ import annotations

import hashlib
import json
import os
from functools import lru_cache
from typing import Dict

from . import thresholds

TAU_PATH = os.path.join(os.path.dirname(__file__), "fixtures", "tau_table_2p1.json")


@lru_cache(maxsize=1)
def _tau_table() -> Dict[int, float]:
    with open(TAU_PATH, "rb") as f:
        raw = f.read()
    sha = hashlib.sha256(raw).hexdigest()
    if sha != thresholds.TAU_FIXTURE_SHA256:
        raise RuntimeError(f"tau_table_2p1.json SHA256 {sha} != {thresholds.TAU_FIXTURE_SHA256}")
    doc = json.loads(raw)
    p = doc["params"]
    if not (p["alpha"] == thresholds.GATE_ALPHA
            and p["mc_seed"] == thresholds.GATE_NULL_MC_SEED
            and p["mc_reps"] == thresholds.GATE_NULL_MC_REPS
            and p["n_max"] == thresholds.GATE_TAU_N_MAX):
        raise RuntimeError(f"tau_table_2p1.json params {p} drifted from thresholds")
    return {int(k): float(v) for k, v in doc["tau"].items()}


def tau(n: int) -> float:
    """tau(n) for n in [2, 256]; RAISES outside (abort, never extrapolate)."""
    table = _tau_table()
    if n in table:
        return table[n]
    raise ValueError(f"tau(n) requested for n={n} outside [2, {thresholds.GATE_TAU_N_MAX}]: "
                     "the 2+1 run must ABORT (comité 052 step 1).")


def abstains(improvement_value: float, n: int) -> bool:
    if n < 2:
        return True
    return improvement_value < tau(n)

=== DOCUMENT 6: scorer.py ===

"""scorer — reveals r to MEASURE where the blind 2+1 boundary fell. Never feeds back.

Same bracket as the sealed 1+1 scorer (nachocausal/scoring/scorer.py), with 2M from the 2+1
thresholds: r_lo = max r over low-O minimal elements, r_hi = min r over high-O ones.

from __future__ import annotations

def blind_bracket(O_by_min: Dict[int, int], min_idx: List[int], thr: float,
                  r: np.ndarray) -> dict:
    lo_r = [float(r[i]) for i in min_idx if O_by_min[i] < thr]
    hi_r = [float(r[i]) for i in min_idx if O_by_min[i] >= thr]
    if not lo_r or not hi_r:
        return dict(valid=False, r_lo=float("nan"), r_hi=float("nan"), width=float("nan"),
                    midpoint=float("nan"), covers=False, clean=False)
    r_lo, r_hi = max(lo_r), min(hi_r)
    width = r_hi - r_lo
    return dict(valid=True, r_lo=r_lo, r_hi=r_hi, width=width, midpoint=0.5 * (r_lo + r_hi),
                covers=bool(r_lo <= thresholds.TWO_M <= r_hi), clean=bool(width >= 0.0))

"""2+1D equatorial Schwarzschild GENERATOR — prereg-004 candidate sub-package (NOT FROZEN).

# z range: w_min ~ (1 - s_max) e^{s_max} * e^{T + R_HI - R_LO}; margin of 1 in z
ZMIN = -4.0

ZT = np.linspace(ZMIN, ZMAX, NZ)
_W = 1.0 - np.exp(ZT)
_S = 1.0 + lambertw(-_W / math.e, 0).real
QT = 2.0 * np.exp(-_S / 2.0) / _S ** 1.5
DLNF = (1.0 + 3.0 / _S) / (_S * np.exp(_S))
DZ = (ZMAX - ZMIN) / (NZ - 1)
# calibration W_c = c ln V + Phi_c(w), Phi_c' = F / (2 (c + sqrt(c^2 + F w))), on a z grid
NZC = 20001
ZC = np.linspace(ZMIN, ZMAX, NZC)
WC = 1.0 - np.exp(ZC)                                     # decreasing in z
_SC = 1.0 + lambertw(-WC / math.e, 0).real
FC = (2.0 * np.exp(-_SC / 2.0) / _SC ** 1.5) ** 2
C_LIST = np.concatenate([np.geomspace(1e-4, 1e3, 400), -np.geomspace(1e-4, 1e3, 200)])


def _phi_table(c):
    """Phi_c(w) on WC (NaN where c is invalid); valid set = one w-interval."""
    disc = c * c + FC * WC
    den = c + np.sqrt(np.maximum(disc, 0.0))
    ok = (disc >= 0.0) & (den > 0.0) & ((c >= 0.0) | (WC > 0.0))
    fp = FC / (2.0 * np.where(ok, den, 1.0))
    out = np.full(WC.shape, np.nan)
    idx = np.nonzero(ok)[0]
    if idx.size < 2 or np.any(np.diff(idx) != 1):
        return out, (np.nan, np.nan)
    i0, i1 = idx[0], idx[-1]
    # integrate in w (WC decreases with index): Phi(w) relative to the largest valid w
    seg = 0.5 * (fp[i0 + 1:i1 + 1] + fp[i0:i1]) * (WC[i0 + 1:i1 + 1] - WC[i0:i1])
    out[i0:i1 + 1] = np.concatenate([[0.0], np.cumsum(seg)])
    return out, (WC[i1], WC[i0])


_PT = [_phi_table(c) for c in C_LIST]
PHI_TAB = np.stack([x[0] for x in _PT])
C_WLO = np.array([x[1][0] for x in _PT]); C_WHI = np.array([x[1][1] for x in _PT])
_keep = np.isfinite(C_WLO)
C_LIST, PHI_TAB, C_WLO, C_WHI = C_LIST[_keep], PHI_TAB[_keep], C_WLO[_keep], C_WHI[_keep]


def point_tables(U, V):

    z = np.log(1.0 - U * V)

        A[k] = c * np.log(V) + np.interp(z, ZC, PHI_TAB[k])
    return A

def to_kruskal(t, r):

def sprinkle(n_mean, rng):
    """Poisson with density ~ r on [0,T] x [R_LO,R_HI] x S^1 (sqrt(-g) = r)."""
    n = rng.poisson(n_mean)
    t = rng.uniform(0.0, T_EDGE, n)
    r = np.sqrt(rng.uniform(R_LO ** 2, R_HI ** 2, n))      # density ~ r
    phi = rng.uniform(0.0, 2.0 * math.pi, n)
    return t, r, phi


@njit(cache=False)
def _tab(T, w):
    z = math.log(1.0 - w)
    x = (z - ZMIN) / DZ
    k = int(x)
    if k < 0:
        k = 0
    if k >= T.size - 1:
        k = T.size - 2
    a = x - k
    return T[k] * (1.0 - a) + T[k + 1] * a


@njit(cache=False)
def qi(w):
    return _tab(QT, w)


@njit(cache=False)
def dlnf(w):
    return _tab(DLNF, w)


@njit(parallel=True, cache=False)
def stage1(U, V, phi, A, cwlo, cwhi):
    n = U.size
    rel = np.zeros((n, n), np.int8)
    for i in prange(n):
        for j in range(n):
            du, dv = U[j] - U[i], V[j] - V[i]
            if not (du > 0.0 and dv > 0.0):
                continue
            d = abs(phi[j] - phi[i])
            if d > math.pi:
                d = 2.0 * math.pi - d
            line = 0.0
            for k in range(GN.size):
                line += GW[k] * qi((U[i] + du * GN[k]) * (V[i] + dv * GN[k]))
            line *= math.sqrt(du * dv)
            # orbit curve: move in U to the hyperbola w = wm < 0, follow it from V_i to V_j,
            # move in U to y; admissible for wm in [w_i, w_j], wm < 0.  Exact angular reach
            # q(wm) sqrt(-wm) ln(V_j / V_i).  Max over N_HYP points of the interval.
            wa, wb = U[i] * V[i], min(U[j] * V[j], -1e-12)
            if wa < wb:
                lnr = math.log(V[j] / V[i])
                for k in range(N_HYP + 1):
                    wm = wa + (wb - wa) * k / N_HYP
                    val = qi(wm) * math.sqrt(-wm) * lnr
                    if val > line:
                        line = val
            if d <= min(line * (1.0 - REL_MARGIN), math.pi) - TOL:
                rel[i, j] = 1
                continue
            # Cauchy-Schwarz with U_b fixed, integrated in y = ln V
            ya, yb = math.log(V[i]), math.log(V[j])
            cs = 0.0
            for k in range(GN.size):
                vv = math.exp(ya + (yb - ya) * GN[k])
                qq = qi(U[j] * vv)
                cs += GW[k] * qq * qq * vv
            ub = math.sqrt(du * cs * (yb - ya))
            if d > min(ub * (1.0 + REL_MARGIN), math.pi) + TOL:
                continue
            # calibration over the c grid valid on the causal rectangle
            w1, w2, w3, w4 = U[i] * V[i], U[i] * V[j], U[j] * V[i], U[j] * V[j]
            wlo = min(min(w1, w2), min(w3, w4)); whi = max(max(w1, w2), max(w3, w4))
            for c in range(cwlo.size):
                if wlo >= cwlo[c] and whi <= cwhi[c]:
                    val = A[c, j] - A[c, i]
                    if val < ub:
                        ub = val
            if d > min(ub * (1.0 + REL_MARGIN), math.pi) + TOL:
                continue
            rel[i, j] = -1
    return rel


@njit(cache=False)
def _shoot(ua, va, vb, p0, steps):
    h = (vb - va) / steps
    U, p, v, L = ua, p0, va, 0.0
    for _ in range(steps):
        k1u = p;                        k1p = dlnf(U * v) * p * (U - v * p)
        u2 = U + 0.5 * h * k1u;         p2 = p + 0.5 * h * k1p;   v2 = v + 0.5 * h
        k2u = p2;                       k2p = dlnf(u2 * v2) * p2 * (u2 - v2 * p2)
        u3 = U + 0.5 * h * k2u;         p3 = p + 0.5 * h * k2p
        k3u = p3;                       k3p = dlnf(u3 * v2) * p3 * (u3 - v2 * p3)
        u4 = U + h * k3u;               p4 = p + h * k3p;         v4 = v + h
        k4u = p4;                       k4p = dlnf(u4 * v4) * p4 * (u4 - v4 * p4)
        if u2 * v2 >= 0.999 or u3 * v2 >= 0.999 or u4 * v4 >= 0.999:
            return math.inf, 0.0
        l1 = qi(U * v) * math.sqrt(max(p, 0.0))
        l2 = qi(u2 * v2) * math.sqrt(max(p2, 0.0))
        l3 = qi(u3 * v2) * math.sqrt(max(p3, 0.0))
        l4 = qi(u4 * v4) * math.sqrt(max(p4, 0.0))
        L += h * (l1 + 2.0 * l2 + 2.0 * l3 + l4) / 6.0
        U += h * (k1u + 2.0 * k2u + 2.0 * k3u + k4u) / 6.0
        p += h * (k1p + 2.0 * k2p + 2.0 * k3p + k4p) / 6.0
        v = v4
        if p <= 0.0 or not math.isfinite(U) or not math.isfinite(p):
            return math.inf, 0.0
    return U, L


@njit(cache=False)
def _shoot_y(ua, va, vb, P0, steps):
    """V-parametrised geodesic in y = ln V:  U' = P,  P' = P + (F'/F)(UV) V P (U - P),
    L = int q(UV) sqrt(P V) dy.  Returns (U at vb, L)."""
    ya, yb = math.log(va), math.log(vb)
    h = (yb - ya) / steps
    U, P, y, L = ua, P0, ya, 0.0
    for _ in range(steps):
        v1 = math.exp(y); v2 = math.exp(y + 0.5 * h); v4 = math.exp(y + h)
        k1u = P;                 k1p = P + dlnf(U * v1) * v1 * P * (U - P)
        u2 = U + 0.5 * h * k1u;  p2 = P + 0.5 * h * k1p
        k2u = p2;                k2p = p2 + dlnf(u2 * v2) * v2 * p2 * (u2 - p2)
        u3 = U + 0.5 * h * k2u;  p3 = P + 0.5 * h * k2p
        k3u = p3;                k3p = p3 + dlnf(u3 * v2) * v2 * p3 * (u3 - p3)
        u4 = U + h * k3u;        p4 = P + h * k3p
        k4u = p4;                k4p = p4 + dlnf(u4 * v4) * v4 * p4 * (u4 - p4)
        if u2 * v2 >= 0.999 or u3 * v2 >= 0.999 or u4 * v4 >= 0.999:
            return math.inf, 0.0
        l1 = qi(U * v1) * math.sqrt(max(P * v1, 0.0))
        l2 = qi(u2 * v2) * math.sqrt(max(p2 * v2, 0.0))
        l3 = qi(u3 * v2) * math.sqrt(max(p3 * v2, 0.0))
        l4 = qi(u4 * v4) * math.sqrt(max(p4 * v4, 0.0))
        L += h * (l1 + 2.0 * l2 + 2.0 * l3 + l4) / 6.0
        U += h * (k1u + 2.0 * k2u + 2.0 * k3u + k4u) / 6.0
        P += h * (k1p + 2.0 * k2p + 2.0 * k3p + k4p) / 6.0
        y += h
        if P <= 0.0 or not math.isfinite(U) or not math.isfinite(P):
            return math.inf, 0.0
    return U, L


@njit(parallel=True, cache=False)
def shoot_pairs_fast(ua, va, ub, vb, steps_y, steps_u):
    """As shoot_pairs, but V-parametrised pairs integrate in ln V (steps_y), U-parametrised
    ones linearly (steps_u); bisection stops once the endpoint is hit to 1e-12."""
    n = ua.size
    L = np.zeros(n); miss = np.zeros(n)
    for k in prange(n):
        useU = (ub[k] - ua[k]) > (vb[k] - va[k])
        if useU:
            x0, y0, y1, target = va[k], ua[k], ub[k], vb[k]
            slope = (target - x0) / (y1 - y0)
        else:
            x0, y0, y1, target = ua[k], va[k], vb[k], ub[k]
            slope = (target - x0) / (math.log(y1) - math.log(y0))      # dU/dy guess
        lo, hi = math.log(slope) - 30.0, math.log(slope) + 30.0
        Xe = math.inf; Lk = 0.0
        for _ in range(90):
            mid = 0.5 * (lo + hi)
            if useU:
                Xe, Lk = _shoot(x0, y0, y1, math.exp(mid), steps_u)
            else:
                Xe, Lk = _shoot_y(x0, y0, y1, math.exp(mid), steps_y)
            if math.isfinite(Xe) and abs(Xe - target) < 1e-12:
                break
            if Xe > target:
                hi = mid
            else:
                lo = mid
        L[k] = Lk
        miss[k] = abs(Xe - target) if math.isfinite(Xe) else math.inf
    return L, miss


@njit(parallel=True, cache=False)
def shoot_pairs(ua, va, ub, vb, steps):
    n = ua.size
    L = np.zeros(n); miss = np.zeros(n)
    for k in prange(n):
        if (ub[k] - ua[k]) > (vb[k] - va[k]):
            x0, y0, y1, target = va[k], ua[k], ub[k], vb[k]
        else:
            x0, y0, y1, target = ua[k], va[k], vb[k], ub[k]
        slope = (target - x0) / (y1 - y0)
        lo, hi = math.log(slope) - 30.0, math.log(slope) + 30.0
        for _ in range(90):
            mid = 0.5 * (lo + hi)
            Xe, _ = _shoot(x0, y0, y1, math.exp(mid), steps)
            if Xe > target:
                hi = mid
            else:
                lo = mid
        Xe, Lk = _shoot(x0, y0, y1, math.exp(0.5 * (lo + hi)), steps)
        L[k] = Lk
        miss[k] = abs(Xe - target) if math.isfinite(Xe) else math.inf
    return L, miss


def bh_order(t, r, phi, rng=None, audit=20000):
    U, V = to_kruskal(t, r)
    t0 = time.time()
    A = point_tables(U, V)
    rel = stage1(U, V, phi, A, C_WLO, C_WHI)
    t1 = time.time() - t0
    amb = rel == -1
    ii, jj = np.nonzero(amb)
    near, n_fine = 0, 0
    coarse_mask = np.zeros(ii.size, bool)
    if ii.size:
        d = np.abs(phi[jj] - phi[ii]); d = np.minimum(d, 2 * math.pi - d)
        Lc, mc = shoot_pairs_fast(U[ii], V[ii], U[jj], V[jj], *COARSE)
        dmax = np.minimum(Lc, math.pi)
        # a coarse miss is never trusted: those pairs go to the fine shot (which must hit)
        far = (np.abs(d - dmax) > COARSE_MARGIN) & (mc < 1e-9)
        coarse_mask = far
        rel[ii[far], jj[far]] = (d[far] <= dmax[far]).astype(np.int8)
        k = np.nonzero(~far)[0]
        n_fine = int(k.size)
        if k.size:
            Lf, mf = shoot_pairs_fast(U[ii[k]], V[ii[k]], U[jj[k]], V[jj[k]], *FINE)
            assert np.all(mf < 1e-9), "fine shooting missed on %d pairs" % int((mf >= 1e-9).sum())
            dm = np.minimum(Lf, math.pi)
            near = int((np.abs(d[k] - dm) < TOL).sum())
            rel[ii[k], jj[k]] = (d[k] <= dm).astype(np.int8)
    order = rel.astype(bool)
    stats = {"n": int(t.size), "relations": int(order.sum()), "stage3_pairs": int(ii.size),
             "stage3_fine_pairs": n_fine, "near_boundary_pairs_tol": near,
             "seconds_stage12": round(t1, 1), "seconds_total": round(time.time() - t0, 1)}
    if rng is not None and audit:
        # (a) pairs decided by stages 1-2; (b) pairs decided by the coarse shot: both vs FINE
        si, sj = np.nonzero((U[None, :] > U[:, None]) & (V[None, :] > V[:, None]) & ~amb)
        ci, cj = ii[coarse_mask], jj[coarse_mask]
        out = {}
        for name, (xi, xj) in (("stage12", (si, sj)), ("coarse_shot", (ci, cj))):
            if xi.size == 0:
                continue
            pick = rng.choice(xi.size, size=min(audit, xi.size), replace=False)
            a, b = xi[pick], xj[pick]
            La, ma = shoot_pairs_fast(U[a], V[a], U[b], V[b], *FINE)
            da = np.abs(phi[b] - phi[a]); da = np.minimum(da, 2 * math.pi - da)
            hit = ma < 1e-9
            ex = da <= np.minimum(La, math.pi)
            out[name] = {"sample": int(pick.size), "endpoint_failures": int((~hit).sum()),
                         "disagreements_among_hit": int((ex[hit] != order[a, b][hit]).sum())}
        stats["audit_vs_fine"] = out
    return order, stats


def mink_order(t, r, phi):
    x, y = r * np.cos(phi), r * np.sin(phi)
    dt = t[None, :] - t[:, None]
    dist = np.hypot(x[None, :] - x[:, None], y[None, :] - y[:, None])
    return (dt > 0.0) & (dt >= dist)


def radial_check(t, r):
    """Same-angle pairs: the 2+1 order must equal the sealed 1+1D EF closed form.
    Returns (mismatches, kruskal_ordered_pairs). A mismatch makes the seed INVALID."""
    from nachocausal.generator import past_matrix_fast
    C = past_matrix_fast(np.c_[t, r], "BH", r_S=TWO_M)          # C[i,j]: j precedes i
    U, V = to_kruskal(t, r)
    K = (U[None, :] > U[:, None]) & (V[None, :] > V[:, None])     # K[i,j]: i precedes j
    return int((K != C.T).sum()), int(K.sum())


def transitivity_violations(order):
    """Number of pairs i<k<j with not i<j. Any violation makes the seed INVALID."""
    A = order.astype(np.float32)
    return int((((A @ A) > 0.5) & ~order).sum())

def test_progress_line_is_blind():

def test_progress_log_written_per_seed_before_the_verdict(monkeypatch, tmp_path):
    monkeypatch.setattr(validate, "RESULTS_DIR", str(tmp_path))
    calls = []

    def fake_per_seed(seed, rho, guard=True):
        path = validate._progress_path("probe")
        n_before = sum(1 for _ in open(path)) if os.path.exists(path) else 0
        calls.append(n_before)
        return dict(_row(), seed=seed, rho=rho, seconds=0.0)

    monkeypatch.setattr(validate, "per_seed", fake_per_seed)
    seeds = list(thresholds.DRY_RUN_SEEDS[:3])
    validate.run(seeds=seeds, label="probe", write=True)
    lines = [json.loads(x) for x in open(validate._progress_path("probe"))]
    assert lines[0]["event"] == "start"
    body = lines[1:]
    assert len(body) == len(seeds) * len(thresholds.RHO_GRID)
    assert all(set(x) == set(validate.PROGRESS_FIELDS) for x in body)
    assert calls == list(range(1, 1 + len(body)))   # each line is on disk before the next seed
    validate.run(seeds=seeds, label="probe", write=False)
    assert sum(1 for _ in open(validate._progress_path("probe"))) == len(lines)  # write=False: none
